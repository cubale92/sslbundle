cmake_minimum_required(VERSION 3.22)
project(Console4KTests LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MPC_VST_PATH "" CACHE PATH "Pinned mpc-vst-plugins checkout")
option(ECEE_SANITIZERS "Enable address and undefined behaviour sanitizers" OFF)
if(ECEE_SANITIZERS AND NOT MSVC)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
include_directories(../shared ../mpc/common)
add_executable(dsp_tests dsp_tests.cpp)
add_test(NAME dsp_tests COMMAND dsp_tests)
if(MPC_VST_PATH)
    foreach(plugin channel4k bus4k listen4k)
        add_executable(${plugin}_engine_tests engine_tests.cpp ../mpc/${plugin}/engine.cpp)
        target_include_directories(${plugin}_engine_tests PRIVATE ${MPC_VST_PATH}/wrapper ../mpc/${plugin})
        if(MSVC)
            target_compile_options(${plugin}_engine_tests PRIVATE /FI${CMAKE_CURRENT_SOURCE_DIR}/msvc_compat.h)
        endif()
        add_test(NAME ${plugin}_engine_tests COMMAND ${plugin}_engine_tests)
    endforeach()
endif()
